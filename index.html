// ========== متغيرات إضافية ========== let currentLedgerCustomerId = ''; // ========== عرض العملاء مع الرصيد وزر السجل ========== function renderCustomersTable(customers) { const tbody = document.getElementById('customersTbody'); if (!tbody) return; const search = (document.getElementById('searchCustomer')?.value || '').toLowerCase(); let html = ''; customers .filter(c => String(c.customer_name || c.name || "").toLowerCase().includes(search)) .forEach(c => { const balance = c.balance || 0; html += ` ${escapeHtml(c.customer_id || c.id || "")} ${escapeHtml(c.customer_name || c.name || "")} ${escapeHtml(c.tax_id || "")} ${escapeHtml(getVendorNo(c))} ${escapeHtml(c.phone || "")} ${escapeHtml(c.address || "")} ${balance.toFixed(2)} `; }); tbody.innerHTML = html; } // ========== عرض سجل الحركات ========== async function viewLedger(customerId, customerName) { currentLedgerCustomerId = customerId; document.getElementById('ledgerModalTitle').innerText = `سجل حركات: ${customerName}`; try { // جلب الرصيد الحالي const balanceData = await apiCall('getCustomerBalance', 'GET', null, { customer_id: customerId }); document.getElementById('ledgerBalance').innerText = (balanceData.balance || 0).toFixed(2); // جلب الدفتر const ledgerData = await apiCall('getCustomerLedger', 'GET', null, { customer_id: customerId }); if (ledgerData.success && ledgerData.ledger) { const tbody = document.getElementById('ledgerTbody'); let html = ''; ledgerData.ledger.forEach(t => { html += ` ${formatDate(t.date)} ${escapeHtml(t.type)} ${escapeHtml(t.reference || "")} ${(t.debit || 0).toFixed(2)} ${(t.credit || 0).toFixed(2)} ${(t.balance || 0).toFixed(2)} ${escapeHtml(t.note || "")} `; }); tbody.innerHTML = html || 'لا توجد حركات'; } else { document.getElementById('ledgerTbody').innerHTML = 'لا توجد حركات'; } // فتح المودال new bootstrap.Modal(document.getElementById('ledgerModal')).show(); } catch (e) { console.error(e); alert("حدث خطأ أثناء تحميل سجل الحركات"); } } // ========== تعديل الرصيد يدوياً ========== document.getElementById('addManualBalanceBtn')?.addEventListener('click', function() { document.getElementById('manualCustomerId').value = currentLedgerCustomerId; document.getElementById('manualAmount').value = ''; document.getElementById('manualNotes').value = ''; new bootstrap.Modal(document.getElementById('manualBalanceModal')).show(); }); document.getElementById('confirmManualBalance')?.addEventListener('click', async function() { const customerId = document.getElementById('manualCustomerId').value; const type = document.getElementById('manualType').value; const amount = document.getElementById('manualAmount').value; const notes = document.getElementById('manualNotes').value; if (!amount || Number(amount) <= 0) { alert('أدخل مبلغ صحيح'); return; } try { const result = await apiCall('addBalanceAdjustment', 'POST', { action: 'addBalanceAdjustment', customer_id: customerId, date: new Date().toISOString().split('T')[0], type: type, amount: Number(amount), notes: notes }); if (result.success) { alert('تم تحديث الرصيد بنجاح'); bootstrap.Modal.getInstance(document.getElementById('manualBalanceModal')).hide(); bootstrap.Modal.getInstance(document.getElementById('ledgerModal')).hide(); // إعادة تحميل العملاء لتحديث الرصيد await loadCustomers(); } else { alert('خطأ: ' + (result.error || 'فشل التحديث')); } } catch (e) { console.error(e); alert('حدث خطأ أثناء التعديل'); } }); // ========== تعديل دالة loadCustomers ========== // استبدل دالة loadCustomers الموجودة بهذه النسخة async function loadCustomers() { try { const data = await apiCall('getCustomers'); const select = document.getElementById('customerSelect'); if (!select) { console.error("customerSelect element not found"); return; } select.innerHTML = ''; if (data.success && Array.isArray(data.customers)) { allCustomers = data.customers; // حفظ البيانات مع الرصيد allCustomers.forEach(c => { const customerId = c.customer_id || c.id || ""; const customerName = c.customer_name || c.name || ""; const option = document.createElement("option"); option.value = customerId; option.textContent = customerName; select.appendChild(option); }); renderCustomersTable(allCustomers); // عرض الرصيد والأزرار } else { console.error("فشل تحميل العملاء:", data); select.innerHTML = ''; } } catch (e) { console.error("خطأ تحميل العملاء:", e); alert("خطأ في تحميل العملاء من الـ API"); } }
